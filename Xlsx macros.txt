Option Explicit

Sub CollectDataFromFolder()
    Dim wbThis As Workbook
    Dim wsNames As Variant
    Dim folderPath As String
    Dim fileName As String
    Dim wbSrc As Workbook
    Dim wsSrc As Worksheet, wsDest As Worksheet
    Dim destRow As Long
    Dim srcLastDataRow As Long
    Dim rngCopy As Range
    Dim i As Long
    Dim foundCell As Range
    Dim lastCol As Long
    Dim numRows As Long

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    Set wbThis = ThisWorkbook

    If wbThis.Path = "" Then
        MsgBox "Пожалуйста, сохраните книгу перед запуском макроса.", vbExclamation, "Ошибка"
        GoTo CleanUp
    End If

    folderPath = wbThis.Path & Application.PathSeparator
    wsNames = Array("2024", "2025", "1 кв 2026")

    fileName = Dir(folderPath & "*.xlsx")

    Do While fileName <> ""
        If StrComp(folderPath & fileName, wbThis.FullName, vbTextCompare) <> 0 Then
            On Error Resume Next
            Set wbSrc = Workbooks.Open(folderPath & fileName, ReadOnly:=True, UpdateLinks:=0)
            On Error GoTo 0

            If Not wbSrc Is Nothing Then
                For i = LBound(wsNames) To UBound(wsNames)
                    Set wsSrc = Nothing
                    On Error Resume Next
                    Set wsSrc = wbSrc.Worksheets(CStr(wsNames(i)))
                    On Error GoTo 0

                    If Not wsSrc Is Nothing Then
                        Set foundCell = FindDoctorRow(wsSrc)
                        If Not foundCell Is Nothing Then
                            srcLastDataRow = foundCell.Row - 1
                        Else
                            srcLastDataRow = wsSrc.Cells(wsSrc.Rows.Count, 1).End(xlUp).Row
                        End If

                        If srcLastDataRow >= 7 Then
                            lastCol = wsSrc.Cells(6, wsSrc.Columns.Count).End(xlToLeft).Column
                            If lastCol < 1 Then lastCol = 1

                            Set rngCopy = wsSrc.Range(wsSrc.Cells(7, 1), wsSrc.Cells(srcLastDataRow, lastCol))
                            numRows = rngCopy.Rows.Count

                            On Error Resume Next
                            Set wsDest = wbThis.Worksheets(CStr(wsNames(i)))
                            On Error GoTo 0

                            If Not wsDest Is Nothing Then
                                Set foundCell = FindDoctorRow(wsDest)
                                If Not foundCell Is Nothing Then
                                    destRow = foundCell.Row
                                Else
                                    destRow = wsDest.Cells(wsDest.Rows.Count, 1).End(xlUp).Row + 1
                                End If

                                ' Безопасная вставка с обработкой объединённых ячеек
                                SafeInsertAndPaste wsDest, destRow, numRows, rngCopy
                            End If
                        End If
                    End If
                Next i
                wbSrc.Close SaveChanges:=False
                Set wbSrc = Nothing
            End If
        End If
        fileName = Dir()
    Loop

    MsgBox "Данные успешно собраны!", vbInformation, "Готово"

CleanUp:
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
End Sub

Private Function FindDoctorRow(ws As Worksheet) As Range
    Dim r As Range
    Dim lastRow As Long
    Dim lastCol As Long

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If lastRow < 7 Then lastRow = 7
    lastCol = ws.Cells(6, ws.Columns.Count).End(xlToLeft).Column
    If lastCol < 1 Then lastCol = 1

    For Each r In ws.Range(ws.Cells(7, 1), ws.Cells(lastRow, lastCol))
        If InStr(1, r.Text, "Главный врач", vbTextCompare) > 0 Then
            Set FindDoctorRow = ws.Rows(r.Row)
            Exit Function
        End If
    Next r
    Set FindDoctorRow = Nothing
End Function

' Версия, работающая во всех версиях Excel (без MergeAreas)
Private Sub SafeInsertAndPaste(wsDest As Worksheet, destRow As Long, numRows As Long, rngCopy As Range)
    Dim mergedDict As Object  ' Dictionary для хранения адресов объединений
    Dim cell As Range
    Dim addr As String
    Dim ma As Range
    Dim i As Long
    Dim rng As Range

    ' 1. Сбор всех уникальных объединённых диапазонов на листе
    Set mergedDict = CreateObject("Scripting.Dictionary")
    On Error Resume Next
    For Each cell In wsDest.UsedRange
        If cell.MergeCells Then
            addr = cell.MergeArea.Address(False, False)
            If Not mergedDict.Exists(addr) Then
                mergedDict.Add addr, cell.MergeArea.Row  ' запоминаем исходную строку начала
            End If
        End If
    Next cell
    On Error GoTo 0

    ' 2. Снимаем объединения (кроме тех, что пересекают destRow)
    Dim keysToRemove As Object
    Set keysToRemove = CreateObject("Scripting.Dictionary")
    Dim key As Variant
    For Each key In mergedDict.Keys
        Set ma = wsDest.Range(CStr(key))
        ' Если диапазон пересекает линию вставки (начало < destRow <= конец), пропускаем
        If ma.Row + ma.Rows.Count - 1 < destRow Or ma.Row >= destRow Then
            ' Сохраним для восстановления
            ma.UnMerge
        Else
            ' Такой диапазон мы не сможем корректно восстановить, поэтому просто снимаем без сохранения
            ma.UnMerge
            keysToRemove.Add key, True   ' удалим из словаря позже
        End If
    Next key
    ' Удаляем из словаря те, что мы не будем восстанавливать
    For Each key In keysToRemove.Keys
        mergedDict.Remove key
    Next key

    ' 3. Вставляем пустые строки перед destRow (теперь ошибок не будет)
    wsDest.Rows(destRow).Resize(numRows).Insert Shift:=xlDown, CopyOrigin:=xlFormatFromLeftOrAbove

    ' 4. Копируем данные из исходного диапазона
    rngCopy.Copy
    wsDest.Cells(destRow, 1).PasteSpecial Paste:=xlPasteAll, Operation:=xlNone, SkipBlanks:=False, Transpose:=False
    Application.CutCopyMode = False

    ' 5. Восстанавливаем объединения с учётом сдвига
    Dim newAddr As String
    For Each key In mergedDict.Keys
        Set ma = wsDest.Range(CStr(key)) ' адрес уже изменился из-за вставки, но сам диапазон пока существует?
        ' После вставки строк адрес объединения сдвинулся, но исходный key – это старый адрес, который уже не актуален.
        ' Нужно восстановить диапазон с теми же относительными координатами, но с учётом сдвига, если исходный диапазон был ниже destRow.
        ' Исходная строка начала сохранена в mergedDict(key).
        Dim origRow As Long
        origRow = CLng(mergedDict(key))
        If origRow >= destRow Then
            ' Сдвиг вниз на numRows
            newAddr = wsDest.Range( _
                wsDest.Cells(origRow + numRows, ma.Column), _
                wsDest.Cells(origRow + ma.Rows.Count - 1 + numRows, ma.Column + ma.Columns.Count - 1) _
            ).Address(False, False)
        Else
            newAddr = key  ' без изменений
        End If
        On Error Resume Next
        wsDest.Range(newAddr).Merge
        On Error GoTo 0
    Next key
End Sub
