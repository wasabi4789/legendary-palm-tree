Sub MergeDataFromFiles()
    Dim SourcePath As String
    Dim SourceFile As String
    Dim TargetWB As Workbook
    Dim SourceWB As Workbook
    Dim wsTarget As Worksheet
    Dim wsSource As Worksheet
    Dim SheetNames As Variant
    Dim i As Integer
    Dim LastRowTarget As Long
    Dim TargetRow As Long
    Dim LastCol As Long
    Dim SourceData As Variant
    Dim rngTarget As Range
    
    ' Текущая книга (принимающая данные)
    Set TargetWB = ThisWorkbook
    SourcePath = TargetWB.Path & "\"
    
    ' Проверка, сохранена ли книга
    If SourcePath = "\" Then
        MsgBox "Книга не сохранена. Сначала сохраните файл.", vbExclamation
        Exit Sub
    End If
    
    ' Список листов для обработки
    SheetNames = Array("2024", "2025", "1 кв 2026")
    
    ' Отключаем обновление экрана и автоматические пересчёты для ускорения
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    
    ' Находим первый файл .xlsx в папке
    SourceFile = Dir(SourcePath & "*.xlsx")
    
    Do While SourceFile <> ""
        ' Пропускаем текущий файл, чтобы не обрабатывать его самого
        If SourceFile <> TargetWB.Name Then
            ' Открываем исходный файл только для чтения
            On Error Resume Next
            Set SourceWB = Workbooks.Open(SourcePath & SourceFile, ReadOnly:=True, UpdateLinks:=False)
            On Error GoTo 0
            
            If Not SourceWB Is Nothing Then
                ' Обрабатываем каждый лист из списка
                For i = LBound(SheetNames) To UBound(SheetNames)
                    ' Проверяем наличие листа в исходной книге
                    On Error Resume Next
                    Set wsSource = SourceWB.Worksheets(SheetNames(i))
                    On Error GoTo 0
                    
                    If Not wsSource Is Nothing Then
                        ' Проверяем наличие листа в целевой книге
                        On Error Resume Next
                        Set wsTarget = TargetWB.Worksheets(SheetNames(i))
                        On Error GoTo 0
                        
                        If Not wsTarget Is Nothing Then
                            ' Определяем последний используемый столбец по 7-й строке (шапка)
                            LastCol = wsSource.Cells(7, wsSource.Columns.Count).End(xlToLeft).Column
                            If LastCol < 1 Then LastCol = 1
                            
                            ' Копируем значения из диапазона A7:LastCol10 в массив
                            SourceData = wsSource.Range(wsSource.Cells(7, 1), wsSource.Cells(10, LastCol)).Value
                            
                            ' Находим последнюю непустую строку в целевом листе (по всем столбцам)
                            On Error Resume Next
                            LastRowTarget = wsTarget.Cells.Find(What:="*", _
                                After:=wsTarget.Cells(1, 1), _
                                LookIn:=xlValues, _
                                LookAt:=xlPart, _
                                SearchOrder:=xlByRows, _
                                SearchDirection:=xlPrevious).Row
                            On Error GoTo 0
                            
                            ' Если лист пуст или только шапка, считаем последней строкой 6
                            If LastRowTarget = 0 Then LastRowTarget = 6
                            If LastRowTarget < 7 Then LastRowTarget = 6
                            
                            ' Определяем строку для вставки: после 13-й строки
                            If LastRowTarget < 13 Then
                                TargetRow = 14
                            Else
                                TargetRow = LastRowTarget + 1
                            End If
                            
                            ' Вставляем значения в целевой лист
                            Set rngTarget = wsTarget.Cells(TargetRow, 1).Resize(4, LastCol)
                            rngTarget.Value = SourceData
                        End If
                        Set wsTarget = Nothing
                    End If
                    Set wsSource = Nothing
                Next i
                
                ' Закрываем исходную книгу без сохранения
                SourceWB.Close SaveChanges:=False
                Set SourceWB = Nothing
            End If
        End If
        ' Переходим к следующему файлу
        SourceFile = Dir
    Loop
    
    ' Восстанавливаем настройки Excel
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    
    MsgBox "Обработка завершена. Данные добавлены в листы 2024, 2025 и 1 кв 2026.", vbInformation
End Sub
